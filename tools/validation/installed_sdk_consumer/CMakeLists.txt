cmake_minimum_required(VERSION 3.24)
project(gpu_nam_installed_sdk_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
foreach(_version IN ITEMS GPU_NAM_EXPECTED_PULP_VERSION GPU_NAM_EXPECTED_VELLUM_VERSION GPU_NAM_EXPECTED_VELLUM_GPU_RUNTIME_VERSION)
  if(NOT ${_version})
    message(FATAL_ERROR "${_version} is required; the parent project passes its own pin")
  endif()
endforeach()
find_package(Pulp ${GPU_NAM_EXPECTED_PULP_VERSION} EXACT REQUIRED CONFIG GLOBAL)
find_package(Vellum ${GPU_NAM_EXPECTED_VELLUM_VERSION} EXACT REQUIRED CONFIG GLOBAL)
find_package(VellumGpuRuntime ${GPU_NAM_EXPECTED_VELLUM_GPU_RUNTIME_VERSION} EXACT REQUIRED CONFIG GLOBAL)
if(NOT TARGET VellumGpuRuntime::DeviceApi OR NOT TARGET VellumGpuRuntime::Device)
  message(FATAL_ERROR "the installed Vellum GPU runtime targets are incomplete")
endif()
add_executable(gpu_nam_installed_sdk_consumer main.cpp)
target_link_libraries(gpu_nam_installed_sdk_consumer PRIVATE Pulp::gpu-audio Vellum::Graphics VellumGpuRuntime::DeviceApi VellumGpuRuntime::Device)
